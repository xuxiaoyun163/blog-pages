二叉堆#

定义#

二叉堆是一棵完全二叉树

类型#

最大堆

  • 父节点的值总是大于或等于其子节点的值
  • 根节点是整个堆中的最大值

最小堆

  • 父节点的值总是小于或等于其子节点的值
  • 根节点是整个堆中的最小值

操作#

二叉堆有两个重要操作一个是插入操作，一个是删除操作

P3378 【模板】堆#

#include<iostream>
#include<vector>
using namespace std;

// 插入操作，将数组维护成最小堆
void push(vector<int> &v)
{
    int i = v.size()-1; // 操作元素的下标

    while( v[i] < v[(i-1)/2] ){
            swap(v[i],v[(i-1)/2]);
            i  = (i -1)/2;
    }
    
}

// 将堆顶元素取出，将数组维护成最小堆
void pop(vector<int> &v){
    if(v.empty()) return;

     v[0] = v.back();
     v.pop_back();// 删除最后一个元素

     if(v.empty()) return;
    int i = 0; //操作元素的下标
    int left = 2*i+1;
    while(left < v.size()){
            int right = left+1;
            int min_index =0;
          if(right >= v.size()){
                min_index = left;  
          }else{
              min_index = v[left]< v[right] ? left : right; 
          }
        // 是否需要交换
        if(v[min_index] < v[i]){
                swap(v[min_index] , v[i]);
                i = min_index;
                left = 2*i+1;
        }else{
            break;
        }
        
    }
        
}
