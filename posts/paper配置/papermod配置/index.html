配置数学公式渲染#

  1. 下载KaTeX 地址：https://cdn.jsdelivr.net/npm/katex@0.16.11/dist/
    我们直接在项目的根目录中执行npm install katex@0.16.11
    安装完成后，node_modules/katex/dist/ 里面就是完整 dist，把里面全部复制到 static/katex/
  2. 修改根目录下的layouts/partials/extend_head.html
{{ if or .Params.math .Site.Params.math }}
<link rel="stylesheet" href="/katex/katex.min.css">
<script defer src="/katex/katex.min.js"></script>
<script defer src="/katex/contrib/auto-render.min.js"></script>
<script>
document.addEventListener("DOMContentLoaded", function() {
    renderMathInElement(document.body,{
      delimiters: [
        {left: '$$', right: '$$', display: true},
        {left: '$', right: '$', display: false}
      ]
    });
});
</script>
{{ end }}
  1. 配置hugo.yml
markup:
  goldmark:
    renderer:
      unsafe: true
    extensions:
      passthrough:
        enable: true
        delimiters:
          block:
            - - "$$"
              - "$$"
          inline:
            - - "$"
              - "$"

配置目录在右侧#

  1. 在根目录下的assets/css/extended/toc.css创建一个toc.css
/* assets/css/extended/toc.css */

.toc-sidebar {
  /* 脱离文档流，不会挤压文章 */
  position: fixed;
  top: 150px;
  right: 200px;
  width: 240px;
  max-height: 85vh;
  overflow-y: auto;
  padding: 1rem;
  z-index: 10;
  /* 可选：增加半透明背景，防止文字重叠看不清 */
  background: var(--theme); // 让背景和主题一个颜色，方便主题颜色切换
  border-radius: 8px;
}

.toc-sidebar .toc {
  background: var(--theme);
  border: 0px solid var(--border);
}

/* 小屏幕直接隐藏这个固定目录 */
@media (max-width: 1200px) {
  .toc-sidebar {
    display: none !important;
  }
}

.toc-sidebar #TableOfContents {
  font-size: 0.9rem;
}

.toc-sidebar #TableOfContents ul {
  padding-left: 1rem;
  list-style: none;   /* 去掉前面的小圆点*/
}

.toc-marker {
  margin-right: 0.3rem;
  opacity: 0.6;
}

.toc-expanded>a {
  font-weight: bold;
}

  <script>
    document.addEventListener('DOMContentLoaded', function() {
      var nav = document.querySelector('.toc-sidebar #TableOfContents');
      if (!nav) return;
      nav.querySelectorAll('li').forEach(function(li) {
        var sub = li.querySelector('ul');
        if (!sub) return;
        var a = li.querySelector('a');
        if (!a) return;
        var level = 0;
        var el = a.parentNode;
        while (el && el !== nav) {
          if (el.tagName === 'UL') level++;
          el = el.parentNode;
        }
         // 会在目录前面加上#号
        // var marker = document.createElement('span');
        // marker.className = 'toc-marker';
        // marker.textContent = '#'.repeat(level + 1);
        // a.parentNode.insertBefore(marker, a);
      });
      nav.querySelectorAll('li').forEach(function(li) {
        var sub = li.querySelector('ul');
        if (!sub) return;
        sub.style.display = 'none';
        li.addEventListener('click', function(e) {
          if (e.target.tagName === 'A') e.preventDefault();
          var isOpen = sub.style.display !== 'none';
          sub.style.display = isOpen ? 'none' : 'block';
          li.classList.toggle('toc-expanded', !isOpen);
        });
      });
    });
  </script>
</aside>
